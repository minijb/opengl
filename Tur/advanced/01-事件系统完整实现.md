## 1. `src/events/Event.h`

```cpp
// events/Event.h —— 事件系统核心:类型枚举 + 分类位掩码 + Event 基类 + EventDispatcher
#pragma once
#include <functional>
#include <ostream>
#include <string>

// ── 事件的分类标签(位掩码,方便一个 handler 订阅多类) ──
enum class EventCategory : int {
    None        = 0,
    Application = 1 << 0,   // 窗口关闭、缩放、焦点、移动
    Input       = 1 << 1,   // 任何输入
    Keyboard    = 1 << 2,
    Mouse       = 1 << 3,
    MouseButton = 1 << 4,
};

// ── 全部事件类型的登记表(新增事件必须在这里登记) ──
enum class EventType {
    None = 0,
    WindowClose, WindowResize, WindowFocus, WindowMoved,
    KeyPressed, KeyReleased, KeyTyped,
    MouseButtonPressed, MouseButtonReleased, MouseMoved, MouseScrolled
};

// 省去每个子类重复写三个一模一样的虚函数
#define EVENT_CLASS_TYPE(type)                                                  \
    static EventType getStaticType() { return EventType::type; }                \
    virtual EventType getEventType() const override { return getStaticType(); } \
    virtual const char* getName() const override { return #type; }

#define EVENT_CLASS_CATEGORY(category) \
    virtual int getCategoryFlags() const override { return static_cast<int>(category); }

// ── 抽象基类:只管类型与分类,不携带具体数据 ──
class Event {
public:
    bool handled = false;   // 任一 handler 置 true,事件视为"已被吃掉",下游可见

    virtual ~Event() = default;

    virtual EventType getEventType() const = 0;
    virtual const char* getName() const = 0;
    virtual int getCategoryFlags() const = 0;
    virtual std::string toString() const { return getName(); }

    bool isInCategory(EventCategory c) const {
        return (getCategoryFlags() & static_cast<int>(c)) != 0;
    }
};

// 统一的回调签名:整条事件管道只认它一个(见设计篇 1.5)。
// 必须定义在 class Event 之后——签名里的 Event& 要求 Event 已声明。
using EventCallback = std::function<void(Event&)>;

// 支持 EventCategory::Keyboard | EventCategory::Input 这类组合
inline int operator|(EventCategory a, EventCategory b) {
    return static_cast<int>(a) | static_cast<int>(b);
}
// 三个及以上分类连写时,左值已是 int,补一个 int | EventCategory 重载
inline int operator|(int a, EventCategory b) {
    return a | static_cast<int>(b);
}

inline std::ostream& operator<<(std::ostream& os, const Event& e) {
    return os << e.toString();
}

// ── 类型安全的派发器:编译期取 T::getStaticType(),匹配才 static_cast ──
class EventDispatcher {
public:
    EventDispatcher(Event& event) : m_event(event) {}

    template<typename T, typename F>
    bool dispatch(const F& func) {
        if (m_event.getEventType() == T::getStaticType()) {
            m_event.handled |= func(static_cast<T&>(m_event));  // |=:一旦 handled 就保持
            return true;
        }
        return false;
    }

private:
    Event& m_event;   // 只存引用,不拷贝事件
};
```

```cpp
#pragma once
#include "Event.h"

// ── 键盘事件中间基类:携带 keyCode ──
class KeyEvent : public Event {
public:
    int getKeyCode() const { return m_keyCode; }

    EVENT_CLASS_CATEGORY(EventCategory::Keyboard | EventCategory::Input)

protected:
    KeyEvent(int keycode) : m_keyCode(keycode) {}
    int m_keyCode;
};

class KeyPressedEvent : public KeyEvent {
public:
    KeyPressedEvent(int keycode, int repeatCount)
        : KeyEvent(keycode), m_repeatCount(repeatCount) {}

    int getRepeatCount() const { return m_repeatCount; }

    std::string toString() const override {
        return "KeyPressed: " + std::to_string(m_keyCode) +
               " (" + std::to_string(m_repeatCount) + " repeats)";
    }

    EVENT_CLASS_TYPE(KeyPressed)

private:
    int m_repeatCount;
};

class KeyReleasedEvent : public KeyEvent {
public:
    KeyReleasedEvent(int keycode) : KeyEvent(keycode) {}

    std::string toString() const override {
        return "KeyReleased: " + std::to_string(m_keyCode);
    }

    EVENT_CLASS_TYPE(KeyReleased)
};

// 字符输入(来自 glfwSetCharCallback 的 Unicode 码点:已处理 Shift/输入法)
class KeyTypedEvent : public KeyEvent {
public:
    KeyTypedEvent(unsigned int codepoint) : KeyEvent(static_cast<int>(codepoint)) {}

    unsigned int getCodepoint() const { return static_cast<unsigned int>(m_keyCode); }

    std::string toString() const override {
        return "KeyTyped: " + std::to_string(m_keyCode);
    }

    EVENT_CLASS_TYPE(KeyTyped)
};
```

## 5. `src/events/EventBus.h`

```cpp
// events/EventBus.h —— 可选中间件:过滤器链 + 最终 handler(设计篇 1.6 / 练习 3)
#pragma once
#include <vector>
#include "Event.h"

// 返回 true = 拦截(吃掉,链路终止);false = 放行
using EventFilter = std::function<bool(Event&)>;

class EventBus {
public:
    void addFilter(EventFilter filter) {
        m_filters.push_back(std::move(filter));
    }

    void setHandler(EventCallback handler) {
        m_final = std::move(handler);
    }

    void dispatch(Event& e) {
        for (auto& filter : m_filters)
            if (filter(e)) return;   // 任一过滤器拦截 → 整条链终止
        if (m_final) m_final(e);     // 全部放行 → 交给最终处理者
    }

private:
    std::vector<EventFilter> m_filters;
    EventCallback            m_final;
};
```

## 6. `src/GLFWManager.h`

```cpp
#pragma once

#include <string>
#include "events/Event.h"

// 前向声明:不在头里暴露 GLFW(见基础篇 03 的 3.5)
struct GLFWwindow;

class GLFWManager {
public:
    // 创建窗口的配置:聚合初始化即可 window.init({"Title", 800, 600})
    struct Props {
        std::string title{"OpenGL App"};
        int  width {800};
        int  height{600};
        bool fullscreen{false};
        bool vsync     {true};
    };

    GLFWManager();
    ~GLFWManager();

    // 窗口是独占资源,禁用拷贝
    GLFWManager(const GLFWManager&)            = delete;
    GLFWManager& operator=(const GLFWManager&) = delete;

    bool init(const Props& props = {});   // 两段式构造,失败返回 false
    void shutdown();

    // —— 主循环 API ——
    void pollEvents();
    void swapBuffers();
    bool shouldClose() const;
    void setShouldClose(bool v);

    // —— 属性 ——
    int  width()  const { return m_width;  }
    int  height() const { return m_height; }
    void setVsync(bool enabled);

    // —— 统一事件通道:替代基础篇的 setKeyCallback / setResizeCallback ——
    // 无论以后加多少种事件,对外始终只有这一个接口
    void setEventCallback(EventCallback cb) { m_eventCb = std::move(cb); }

    // —— 输入查询(轮询式,"持续按住"用这个,见设计篇 1.8) ——
    bool isKeyPressed(int key) const;

    // 逃生舱口:仅在需要原生句柄(如 ImGui)时用
    GLFWwindow* nativeHandle() const { return m_window; }

private:
    // GLFW 回调是 C 函数指针,必须 static;靠 user pointer 找回 this
    static void keyCallback(GLFWwindow* w, int key, int scancode, int action, int mods);
    static void charCallback(GLFWwindow* w, unsigned int codepoint);
    static void cursorPosCallback(GLFWwindow* w, double xpos, double ypos);
    static void mouseButtonCallback(GLFWwindow* w, int button, int action, int mods);
    static void scrollCallback(GLFWwindow* w, double xoffset, double yoffset);
    static void windowCloseCallback(GLFWwindow* w);
    static void framebufferSizeCallback(GLFWwindow* w, int width, int height);
    static void windowFocusCallback(GLFWwindow* w, int focused);
    static void windowPosCallback(GLFWwindow* w, int xpos, int ypos);

    GLFWwindow*   m_window{nullptr};
    int           m_width{0};
    int           m_height{0};
    EventCallback m_eventCb;   // 只剩这一个回调成员(基础篇是 N 个)
};
```

## 7. `src/GLFWManager.cpp`

```cpp
#include "GLFWManager.h"

#include <GL/glew.h>
#include <GLFW/glfw3.h>
#include <iostream>

#include "events/AppEvent.h"
#include "events/KeyEvent.h"
#include "events/MouseEvent.h"

static void glfwErrorCallback(int code, const char* desc) {
    std::cerr << "[GLFW Error " << code << "] " << desc << "\n";
}

GLFWManager::GLFWManager() = default;

GLFWManager::~GLFWManager() { shutdown(); }

bool GLFWManager::init(const Props& props) {
    // (1) glfwInit 是进程级的,整个程序只调一次(见基础篇 04 / 进阶篇 06)
    static bool glfwInited = false;
    if (!glfwInited) {
        glfwSetErrorCallback(glfwErrorCallback);
        if (!glfwInit()) {
            std::cerr << "Failed to initialize GLFW\n";
            return false;
        }
        glfwInited = true;
    }

    // (2) 声明 OpenGL 版本与 profile
    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 4);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 1);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
#ifdef __APPLE__
    glfwWindowHint(GLFW_OPENGL_FORWARD_COMPAT, GL_TRUE);
#endif

    // (3) 创建窗口
    GLFWmonitor* monitor = props.fullscreen ? glfwGetPrimaryMonitor() : nullptr;
    m_window = glfwCreateWindow(props.width, props.height,
                                props.title.c_str(), monitor, nullptr);
    if (!m_window) {
        std::cerr << "Failed to create GLFW window\n";
        return false;
    }

    // (4) 桥接:C 回调 → user pointer → this(见设计篇 1.7.6)
    glfwSetWindowUserPointer(m_window, this);

    // (5) 登记全部 GLFW 回调 —— 每个回调只做一件事:翻译成 Event 喂进统一通道
    glfwSetKeyCallback(m_window, keyCallback);
    glfwSetCharCallback(m_window, charCallback);
    glfwSetCursorPosCallback(m_window, cursorPosCallback);
    glfwSetMouseButtonCallback(m_window, mouseButtonCallback);
    glfwSetScrollCallback(m_window, scrollCallback);
    glfwSetWindowCloseCallback(m_window, windowCloseCallback);
    glfwSetFramebufferSizeCallback(m_window, framebufferSizeCallback);
    glfwSetWindowFocusCallback(m_window, windowFocusCallback);
    glfwSetWindowPosCallback(m_window, windowPosCallback);

    // (6) 激活上下文后才能加载 GL 函数指针
    glfwMakeContextCurrent(m_window);

    static bool glewInited = false;
    if (!glewInited) {
        glewExperimental = GL_TRUE;
        if (glewInit() != GLEW_OK) {
            std::cerr << "Failed to initialize GLEW\n";
            shutdown();
            return false;
        }
        glewInited = true;
    }

    // (7) 记录尺寸、设 VSync
    m_width  = props.width;
    m_height = props.height;
    setVsync(props.vsync);
    return true;
}

void GLFWManager::shutdown() {
    if (m_window) {
        glfwDestroyWindow(m_window);   // 只销毁窗口;glfwTerminate 留给最外层
        m_window = nullptr;
    }
}

void GLFWManager::pollEvents()           { glfwPollEvents(); }
void GLFWManager::swapBuffers()          { glfwSwapBuffers(m_window); }
bool GLFWManager::shouldClose() const    { return glfwWindowShouldClose(m_window); }
void GLFWManager::setShouldClose(bool v) { glfwSetWindowShouldClose(m_window, v); }
void GLFWManager::setVsync(bool enabled) { glfwSwapInterval(enabled ? 1 : 0); }

bool GLFWManager::isKeyPressed(int key) const {
    return m_window && glfwGetKey(m_window, key) == GLFW_PRESS;
}

// ── 静态回调:每个都是「取回 this → 构造 Event → 喂给 m_eventCb」三步 ──

void GLFWManager::keyCallback(GLFWwindow* w, int key, int /*scancode*/, int action, int /*mods*/) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;

    switch (action) {
        case GLFW_PRESS:   { KeyPressedEvent  e(key, 0); self->m_eventCb(e); break; }
        case GLFW_RELEASE: { KeyReleasedEvent e(key);    self->m_eventCb(e); break; }
        case GLFW_REPEAT:  { KeyPressedEvent  e(key, 1); self->m_eventCb(e); break; }
    }
}

void GLFWManager::charCallback(GLFWwindow* w, unsigned int codepoint) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;
    KeyTypedEvent e(codepoint);
    self->m_eventCb(e);
}

void GLFWManager::cursorPosCallback(GLFWwindow* w, double xpos, double ypos) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;
    MouseMovedEvent e(static_cast<float>(xpos), static_cast<float>(ypos));
    self->m_eventCb(e);
}

void GLFWManager::mouseButtonCallback(GLFWwindow* w, int button, int action, int /*mods*/) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;

    switch (action) {
        case GLFW_PRESS:   { MouseButtonPressedEvent  e(button); self->m_eventCb(e); break; }
        case GLFW_RELEASE: { MouseButtonReleasedEvent e(button); self->m_eventCb(e); break; }
    }
}

void GLFWManager::scrollCallback(GLFWwindow* w, double xoffset, double yoffset) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;
    MouseScrolledEvent e(static_cast<float>(xoffset), static_cast<float>(yoffset));
    self->m_eventCb(e);
}

void GLFWManager::windowCloseCallback(GLFWwindow* w) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self) return;
    WindowCloseEvent e;
    if (self->m_eventCb) self->m_eventCb(e);
    // GLFW 在调用本回调之前已把 close 标志置位;
    // 若业务在 handler 里 e.handled = true,视为「不许关」,撤销关闭:
    if (e.handled) glfwSetWindowShouldClose(w, GLFW_FALSE);
}

void GLFWManager::framebufferSizeCallback(GLFWwindow* w, int width, int height) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self) return;
    self->m_width  = width;    // 同步尺寸,否则 width()/height() 过时(基础篇 04 的坑)
    self->m_height = height;
    if (self->m_eventCb) {
        WindowResizeEvent e(width, height);
        self->m_eventCb(e);
    }
}

void GLFWManager::windowFocusCallback(GLFWwindow* w, int focused) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;
    WindowFocusEvent e(focused == GLFW_TRUE);
    self->m_eventCb(e);
}

void GLFWManager::windowPosCallback(GLFWwindow* w, int xpos, int ypos) {
    auto* self = static_cast<GLFWManager*>(glfwGetWindowUserPointer(w));
    if (!self || !self->m_eventCb) return;
    WindowMovedEvent e(xpos, ypos);
    self->m_eventCb(e);
}
```

## 8. `src/main.cpp`

```cpp
// main.cpp —— 事件系统示范:过滤器链 + Dispatcher 分流
#include <GL/glew.h>
#include <GLFW/glfw3.h>
#include <iostream>

#include "GLFWManager.h"
#include "events/AppEvent.h"
#include "events/EventBus.h"
#include "events/KeyEvent.h"
#include "events/MouseEvent.h"

int main() {
    GLFWManager window;
    if (!window.init({ "Event System Demo", 1280, 720 })) {
        std::cerr << "init failed\n";
        return 1;
    }
    std::cout << "window " << window.width() << "x" << window.height() << "\n";

    EventBus bus;

    // 过滤器 1:全局日志(演示 isInCategory 分类位掩码),永不拦截
    bus.addFilter([](Event& e) {
        std::cout << "[log]" << (e.isInCategory(EventCategory::Input) ? " [input]" : "")
                  << " " << e << "\n";
        return false;
    });

    // 过滤器 2:吃掉 WindowMovedEvent —— 演示「拦截」语义。
    // 注意它排在日志过滤器之后:事件先被记录、再被吃掉,最终 handler 收不到。
    bus.addFilter([](Event& e) {
        EventDispatcher d(e);
        return d.dispatch<WindowMovedEvent>([](WindowMovedEvent&) { return true; });
    });

    // 最终 handler:用 Dispatcher 分流到各业务逻辑
    bus.setHandler([&](Event& e) {
        EventDispatcher d(e);

        d.dispatch<WindowResizeEvent>([&](WindowResizeEvent& ev) {
            glViewport(0, 0, ev.getWidth(), ev.getHeight());
            return false;                      // 不吃掉:其它系统还能看到 resize
        });

        d.dispatch<KeyPressedEvent>([&](KeyPressedEvent& ev) {
            if (ev.getKeyCode() == GLFW_KEY_ESCAPE) {
                window.setShouldClose(true);   // ESC 关窗
                return true;
            }
            return false;
        });

        d.dispatch<KeyTypedEvent>([](KeyTypedEvent& ev) {
            std::cout << "typed '" << static_cast<char>(ev.getCodepoint()) << "'\n";
            return false;
        });

        d.dispatch<MouseButtonPressedEvent>([](MouseButtonPressedEvent& ev) {
            std::cout << "click button " << ev.getButton() << "\n";
            return false;
        });
    });

    // 窗口的唯一出口:所有事件先过过滤器链,再到 handler
    window.setEventCallback([&](Event& e) { bus.dispatch(e); });

    while (!window.shouldClose()) {
        glClearColor(0.10f, 0.14f, 0.18f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);
        window.swapBuffers();
        window.pollEvents();   // 事件在 poll 里被翻译并派发(设计篇 1.7.1 的第 2~8 步)
    }

    window.shutdown();
    std::cout << "clean exit\n";
    return 0;
}
```
